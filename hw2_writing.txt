findValue(i, n)
    if n > 1 and i == 2 ^ (n - 1)
        return 1
    else if i == 1
        return 0
    else if(i > 2 ^ (n - 1))
        return 1 - findValue(2 ^ n - i, n - 1)
    else 
        return findValue(i, n - 1)

findTotal(n, k)
    if n == 1 or k == 0
        return 0
    else
        if k < 2 ^ (n - 1) // node k is on the left subtree
            return findTotal(n - 1, k)
        else if k > 2 ^ (n - 1) // node k is on the right subtree
            return k - 2 ^ (n - 1) + findTotal(n - 1, 2 ^ n - k - 1) + 1
        else
            return findTotal(n - 1, 2 ^ (n - 1) - 1) + 1

subpathXOR(k, n, inv)

        value = 0
    else 
        value = 1

    if inv == 1
        value = 1 - value
    
    if k == 2 ^ (n - 1)
        return value
    else if k < 2 ^ (n - 1)
        return value XOR subpathXOR(k, n - 1, inv)
    else
        return value XOR subpathXOR(2 ^ n - k, n - 1, 1 - inv)

processingQueue(processList, Q0, Q1)
    completionTimeArray = createArray(len = n) // records the completion time of ecah process
    for i from 1 to n
        while not isEmpty(Q0) and processList[i].t >= Q0.peek() // strictly increasing order of arrival time
            Q0.dequeue() 
        while not isEmpty(Q1) and processList[i].t >= Q1.peek()
            Q1.dequeue()

        activeQueue = Q0
        if size(Q1) < size(Q0)
            activeQueue = Q1
        
        completionTime = processList[i].c
        if not isEmpty(activeQueue)
            completionTime += activeQueue.getLastElement()
        else
            completionTime += processList[i].t
        completionTimeArray[i] = completionTime
        activeQueue.enqueue(completionTime)
        

    return completionTimeArray

monotonicSort(array A, len n)
    indexArray = createArray(len = n)
    top = 0
    for i from 1 to n
        while top > 0 and A[i] >= A[indexArray[top]]
            top--
        top++
        indexArray[top] = i

    return indexArray

maxScore(array A, indexArray, left, right, top)
    leftPointer = 1, rightPointer = top
    bestTarget = indexArray[top]
    while leftPointer <= rightPointer
        mid = (leftPointer + rightPointer) / 2
        if left <= indexArray[mid]
            bestTarget = indexArray[mid]
            rightPointer = mid - 1

        else 
            leftPointer = mid + 1

    return A[bestTarget]

findMaxScores(array A, len n, queries)
    k = queries.len // assume "queries" is an array with a length of k that contains k queries
    indexArray = createArray(len = n)
    result = createArray(len = k)
    top = 0
    j = 1
    for i from 1 to n
        while top > 0 and A[i] >= A[indexArray[top]]
            top--
        top++
        indexArray[top] = i
        while j <= k && i == queries[j].right
            result[j] = maxScore(A, indexArray, queries[j].left, queries[j].right, top)
            j++

    return result
